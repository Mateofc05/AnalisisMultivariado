Estadístico de prueba de Hotelling: \(t^2 = n (\bar{X} - \mu_0)^T S^{-1} (\bar{X} - \mu_0)\)

def singleHotellingsT2Test(datos: pd.DataFrame, media_nula: np.array,
                           nivel_alpha: float = 0.05) -> float:
    """
    Aplica el contraste T² de Hotelling a una sola muestra multivariada.

    Parámetros
    ----------
    datos: pd.DataFrame
        Observaciones de las variables que se desean contrastar.
    media_nula: np.array
        Vector de medias establecido por la hipótesis nula.
    nivel_alpha: float
        Probabilidad máxima permitida para el error tipo I.
    """
    total_observaciones = len(datos)
    media_muestral = datos.mean().values
    matriz_covarianza = np.cov(datos, rowvar=False)
    total_variables = len(media_nula)
    diferencia_medias = media_muestral - media_nula

    t_cuadrada = (total_observaciones * diferencia_medias.T
                  @ np.linalg.inv(matriz_covarianza) @ diferencia_medias)
    estadistico_f = ((total_observaciones - total_variables) /
                     ((total_observaciones - 1) * total_variables) * t_cuadrada)
    valor_critico_f = stats.f.ppf(
        1 - nivel_alpha, total_variables, total_observaciones - total_variables
    )
    valor_p = 1 - stats.f.cdf(
        estadistico_f, total_variables, total_observaciones - total_variables
    )

    # Mostramos el estadístico obtenido junto con los valores que permiten tomar la decisión.
    print(f"Estadística de prueba: {estadistico_f}\nGrados de libertad: {total_variables}, {total_observaciones + total_variables}\nregion de rechazo: {valor_critico_f}\nnivel de significancia: {nivel_alpha}")

    if estadistico_f > valor_critico_f:
        print("Se rechaza la hipótesis nula: las medias son diferentes.")
    else:
        print("No se rechaza la hipótesis nula: no hay evidencia suficiente para afirmar que las medias son diferentes.")

    return estadistico_f, valor_critico_f, valor_p

def multiple_ttest_ind(grupo_base: pd.DataFrame, grupo_comparado: pd.DataFrame,
                       nivel_alpha: float = 0.05) -> pd.DataFrame:
    
    '''
    Compara dos grupos mediante una prueba t independiente por cada variable.

    Parámetros
    ----------
    grupo_base: pd.DataFrame
        Variables numéricas del grupo usado como referencia.
    grupo_comparado: pd.DataFrame
        Variables numéricas del grupo que se contrasta.
    nivel_alpha: float
        Umbral utilizado para decidir si se rechaza H0.

    Output
    ------
    tabla_resultados: pd.DataFrame
        Resumen del estadístico, valor p y decisión para cada columna.
    '''
    filas_resultado = []
    for variable in grupo_base.columns:
        estadistico_t, valor_p = stats.ttest_ind(
            grupo_base[variable], grupo_comparado[variable], equal_var=False
        )
        rechazar_h0 = valor_p < nivel_alpha
        filas_resultado.append({
            'Variable': variable,
            't-statistic': estadistico_t,
            'p-value': valor_p,
            'Reject H0': rechazar_h0
        })
    
    return pd.DataFrame(filas_resultado)

def bonferroni(datos: pd.DataFrame, nivel_alpha: float = 0.05) -> pd.DataFrame:
    '''
    Construye intervalos simultáneos aplicando el ajuste de Bonferroni.

    Parámetros
    ----------
    datos: pd.DataFrame
        Conjunto de variables numéricas por analizar.
    nivel_alpha: float
        Nivel de significancia conjunto.

    Output
    ------
    intervalos: pd.DataFrame
        Tabla con la media y los extremos del intervalo para cada variable.
    '''
    # Identificamos el tamaño de la muestra y el número de comparaciones involucradas.
    numero_filas, numero_variables = datos.shape
    nombres_variables = datos.columns

    # Estimamos la media y la varianza de cada variable.
    medias = datos.mean()
    varianzas = datos.var()

    # Ajustamos el cuantil de la distribución t según el número de variables.
    cuantil_t = stats.t.ppf(
        1 - nivel_alpha / (2 * numero_variables), numero_filas - 1
    )

    # Usamos el error estándar para obtener los límites del intervalo.
    limites_inferiores = medias - cuantil_t * np.sqrt(varianzas / numero_filas)
    limites_superiores = medias + cuantil_t * np.sqrt(varianzas / numero_filas)

    # Reunimos las estimaciones y sus límites en una tabla de resultados.
    intervalos = pd.DataFrame({
        "variable": nombres_variables,
        "mean": medias,
        "lower": limites_inferiores,
        "upper": limites_superiores
    })

    return intervalos

Estadístico de prueba de Hotelling: \(t^2 = n (\bar{X_1} - \bar{X_2})^T [S_p(\frac{1}{n}+\frac{1}{m})] (\bar{X_1} - \bar{X_2})\)

\(S_p = \frac{(n-1)S_1 + (m-1)S_2}{n + m - 2}\)

def doubleHotellingsT2Test(grupo_uno: pd.DataFrame, grupo_dos: pd.DataFrame,
                           nivel_alpha: float = 0.05) -> tuple[np.float64, np.float64]:
    '''
    Evalúa la igualdad de dos vectores de medias con la prueba T² de Hotelling.

    Parámetros
    ----------
    grupo_uno: pd.DataFrame
        Observaciones correspondientes a la primera muestra.
    grupo_dos: pd.DataFrame
        Observaciones correspondientes a la segunda muestra.

    Output
    ------
    estadistico_f: np.float64
        Transformación de T² que sigue una distribución F.
    valor_p: np.float64
        Probabilidad asociada con el resultado observado.
    '''
    # Guardamos el número de observaciones de cada grupo y la cantidad de variables.
    filas_uno, numero_variables = grupo_uno.shape
    filas_dos = grupo_dos.shape[0]

    # Combinamos las matrices de covarianza ponderándolas por sus grados de libertad.
    diferencia_medias = grupo_uno.mean().values - grupo_dos.mean().values
    covarianza_uno = np.cov(grupo_uno, rowvar=False)
    covarianza_dos = np.cov(grupo_dos, rowvar=False)
    covarianza_combinada = (
        (filas_uno - 1) * covarianza_uno + (filas_dos - 1) * covarianza_dos
    ) / (filas_uno + filas_dos - 2)

    # Obtenemos T² y lo transformamos en el estadístico F correspondiente.
    factor_tamanos = 1 / filas_uno + 1 / filas_dos
    t_cuadrada = (diferencia_medias
                  @ np.linalg.inv(covarianza_combinada * factor_tamanos)
                  @ diferencia_medias)
    estadistico_f = (
        (filas_uno + filas_dos - numero_variables - 1) /
        (numero_variables * (filas_uno + filas_dos - 2)) * t_cuadrada
    )

    # Calculamos el valor p utilizando la distribución F del contraste.
    grados_libertad_dos = filas_uno + filas_dos - numero_variables - 1
    valor_p = 1 - stats.f.cdf(
        estadistico_f, numero_variables, grados_libertad_dos
    )

    # Presentamos el resultado y determinamos la decisión a partir del valor p.
    print(f"Estadística de prueba: {estadistico_f}\nGrados de libertad: {numero_variables}, {grados_libertad_dos}\np-value: {valor_p}\nnivel de significancia: {nivel_alpha}")

    if valor_p < nivel_alpha:
        print("Se rechaza la hipótesis nula: las medias son diferentes.")
    else:
        print("No se rechaza la hipótesis nula: no hay evidencia suficiente para afirmar que las medias son diferentes.")

    return estadistico_f, valor_p
